#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdbool.h>


int primenumber(int range){
    int num;
    int i;
    for(num = 2; num <= range; num++){
        bool isPrime = true;

        for(i = 2; i*i <= num; i++){
            if (num % i == 0) {
                isPrime = false;
                break;
            }}
        if (isPrime){
            printf("%d ", num);
        }
    }
}
