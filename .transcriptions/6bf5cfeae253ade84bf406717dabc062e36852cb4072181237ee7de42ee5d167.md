int main(){
    int status;
    pid_t child = fork();

    if(child == 0){
        int nb =0;
        int i;
        int factorial = 1;
        printf("Enter factorial : ");
        scanf("%d", &nb);
        for(i=1; i<=nb; i++){
            factorial = factorial*i;
        }
        printf("\nfactorial of %d is %d\n", nb, factorial);
        exit(0);
    }
    waitpid(child, &status, 0);

    int range;
    printf("Enter a range for prime number : ");
    scanf("%d", &range);
    primenumber(range);



    return 0;
}
